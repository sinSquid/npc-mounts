cmake_minimum_required(VERSION 3.21)
project(HookInstallTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(CommonLibSSE CONFIG REQUIRED)
find_package(directxtk CONFIG REQUIRED)
add_executable(hook-install ../hook-install.cpp)
target_link_libraries(hook-install PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_options(hook-install PRIVATE /UNDEBUG /utf-8 /EHsc)
add_executable(diagnostics ../diagnostics.cpp)
target_link_libraries(diagnostics PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_options(diagnostics PRIVATE /UNDEBUG /utf-8 /EHsc)
