cmake_minimum_required(VERSION 3.21)
project(NPCMounts VERSION 0.4.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CommonLibSSE CONFIG REQUIRED)
find_package(directxtk CONFIG REQUIRED)
add_library(npcmounts_minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(npcmounts_minhook PUBLIC third_party/minhook/include)
add_commonlibsse_plugin(NPCMounts NAME NPCMounts AUTHOR "npc-mounts" STRUCT_DEPENDENT COMPATIBLE_RUNTIMES 1.5.97 SOURCES src/plugin.cpp src/Config.cpp src/Seat.cpp src/Extras.cpp src/Crawl.cpp src/Travel.cpp src/Wardrobe.cpp)
target_link_libraries(NPCMounts PRIVATE npcmounts_minhook)
target_compile_options(NPCMounts PRIVATE /EHsc /utf-8)
target_precompile_headers(NPCMounts PRIVATE src/pch.h)
