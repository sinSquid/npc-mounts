cmake_minimum_required(VERSION 3.21)
project(NPCMounts VERSION 0.1.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CommonLibSSE CONFIG REQUIRED)
find_package(directxtk CONFIG REQUIRED)
add_commonlibsse_plugin(NPCMounts NAME NPCMounts AUTHOR "npc-mounts" STRUCT_DEPENDENT COMPATIBLE_RUNTIMES 1.5.97 SOURCES src/plugin.cpp)
target_compile_options(NPCMounts PRIVATE /EHsc /utf-8)
target_precompile_headers(NPCMounts PRIVATE src/pch.h)
